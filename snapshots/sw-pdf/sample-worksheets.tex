\begin{theorem}{Theorem}{Ohms Law.}{}{worksheet-networks-2-3}%
The current through a resistor is proportional to the ratio of the \emph{Voltage} to the \emph{Resistance}%
\begin{equation*}
I = \frac{V}{R} 
\end{equation*}
Or for our purposes%
\begin{equation*}
I R = V 
\end{equation*}
%
\end{theorem}
\begin{theorem}{Theorem}{Kirchoffs Current Law.}{}{worksheet-networks-2-4}%
The sum of the currents in a network meeting at a point is zero.%
\begin{equation*}
\sum_{k=1}^n I_k = 0 
\end{equation*}
%
\end{theorem}
\begin{example}{Example}{Kirchoff's Current Law.}{worksheet-networks-2-5}%
For the circuit below \(I_1 + I_2 = I_3 \).%
\begin{image}{0.3}{0.4}{0.3}{}%

to[R, i=$I_1$, *-o](2,2);
\draw (0,0)
to[R, i=$I_2$, *-o](2,-2);
\draw (-3, 0)
to[R, i=$I_3$, o-*](0,0);
\end{circuitikz}%
}%
\end{image}%
\end{example}
\begin{theorem}{Theorem}{Kirchoffs Voltage Law.}{}{worksheet-networks-2-6}%
The sum of the voltages around any closed circuit (or subcircuit) is zero.%
\begin{equation*}
\sum_{k=1}^n V_k = 0 
\end{equation*}
%
\end{theorem}
Kirchoffs Current Law and Kirkoffs Voltage Law combined with Ohms Law gives for any circuit of resistors and sources a linear system that may (or may not) determine the currents.%

\begin{sectionptx}{Section}{Handouts}{}{Handouts}{}{}{section-handouts}
\begin{introduction}{}%
Like worksheets, a \ptxterminology{handout} is a division that is inteded to be printed for use in a classroom. In HTML output, you get the same print preview and page layout as you do with worksheets; in PDF, these will start on a new page with possibly different margins than the rest of the document.%
\par
Unlike worksheets, handouts do not have a special class of exercises or activities (exercises in a handout are treated like an \emph{inline} exercise). The other main distinction is that a handout lets workspace be specified on pretty much any block or paragraph element, not just exercises, tasks, and project-like elements.  This allows them to be used in the creation of guided notes containing some premade content with lots of space for students to fill in details during class.%
\end{introduction}%

\par

\end{dlimedium}%
