             self.assertEqual(add(2,2), 4, "Adding two identical integers")
             self.assertEqual(add(7,13), 20, "Adding two different integers")
             self.assertAlmostEqual(add(2.0,3.0), 5.0, 5, "Adding two decimal numbers")
